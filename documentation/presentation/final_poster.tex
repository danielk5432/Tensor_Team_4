% ── Sec 2: CP Decomposition ───────────────────────────────────────────
\begin{tcolorbox}[sbox, title={2.\;\; CP Decomposition (Canonical Polyadic)}]
  \fontsize{28}{36}\selectfont\RaggedRight
  Sum of $R$ rank-one tensors:
  \vspace{4pt}
  \begin{tcolorbox}[eqbox]
    \fontsize{44}{54}\selectfont\centering
    $\displaystyle\mathcal{X}\approx\sum_{r=1}^{R}
      \mathbf{a}_r\otimes\mathbf{b}_r\otimes\mathbf{c}_r$
  \end{tcolorbox}
  \vspace{8pt}
  \textbf{ALS mode-1 update:}
  \vspace{4pt}
  \begin{tcolorbox}[eqbox]
    \fontsize{34}{42}\selectfont\centering
    $A\leftarrow X_{(1)}(C\odot B)
      (B^{\!\top}B\ast C^{\!\top}C)^{\dagger}$
  \end{tcolorbox}
  \vspace{8pt}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Fix $B,C$ $\to$ update $A$; cycle all modes
    \item Rank $R$ directly controls compression
  \end{itemize}
  \vspace{6pt}
\begin{center}
  \includegraphics[width=0.75\linewidth]{figures/fig1_rank_vs_error.png}
\end{center}
{\fontsize{24}{30}\selectfont\centering
    CP-ALS matches TensorLy: relative error vs.\ rank $R$\par}
\end{tcolorbox}

\vsep

% ── Sec 3: Tucker Decomposition ───────────────────────────────────────
\begin{tcolorbox}[sbox, title={3.\;\; Tucker Decomposition (HOSVD / HOOI)}]
  \fontsize{28}{36}\selectfont\RaggedRight
  Core tensor $\mathcal{G}$ + factor matrices:
  \vspace{4pt}
  \begin{tcolorbox}[eqbox]
    \fontsize{44}{54}\selectfont\centering
    $\mathcal{X}\approx\mathcal{G}
      \times_1\!A\;\times_2\!B\;\times_3\!C$
  \end{tcolorbox}
  \vspace{8pt}
  \textbf{HOSVD init \& HOOI refinement:}
  \vspace{4pt}
  \begin{tcolorbox}[eqbox]
    \fontsize{32}{40}\selectfont\centering
    $A^{(n)}=\mathrm{trunc\text{-}SVD}_{r_n}(X_{(n)})$\quad
    $\mathcal{G}=\mathcal{X}
      \times_1\!A^{\!\top}\times_2\!B^{\!\top}\times_3\!C^{\!\top}$
  \end{tcolorbox}
  \vspace{8pt}
  \begin{itemize}\setlength\itemsep{4pt}
    \item Multi-linear rank $(r_1,r_2,r_3)$ — more flexible than CP
    \item HOSVD: closed-form init;\; HOOI: iterative refinement
  \end{itemize}
  \vspace{6pt}

\end{center}
  {\fontsize{24}{30}\selectfont\centering
    Tucker (Lena): PSNR \& compression ratio vs.\ rank\par}
\end{tcolorbox}

% ── Sec 4: Mathematical Analysis ─────────────────────────────────────
% Layout: [objective eq | metrics eq] / explanation / [fig | fig]
\begin{tcolorbox}[sbox, title={4.\;\; Mathematical Analysis \& Validation}]

  % Row 1 — equations side by side
  \begin{minipage}[t]{0.48\linewidth}
    \fontsize{26}{34}\selectfont\RaggedRight
    \textbf{Optimization objective:}
    \vspace{4pt}
    \begin{tcolorbox}[eqbox]
      \fontsize{30}{38}\selectfont\centering
      $\displaystyle\min_{A,B,C}
        \!\left\|\mathcal{X}-\sum_{r=1}^{R}
          \mathbf{a}_r\!\otimes\!\mathbf{b}_r\!\otimes\!\mathbf{c}_r
        \right\|_F^{\!2}$
    \end{tcolorbox}
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.49\linewidth}
    \fontsize{26}{34}\selectfont\RaggedRight
    \textbf{Evaluation metrics:}
    \vspace{4pt}
    \begin{tcolorbox}[greenbox]
      \fontsize{22}{28}\selectfont
      \begin{align*}
        \varepsilon_{\mathrm{rel}}
          &=\frac{\|\mathcal{X}-\hat{\mathcal{X}}\|_F}{\|\mathcal{X}\|_F}\\[3pt]
        \mathrm{PSNR}
          &=10\log_{10}\!\left(\frac{\mathrm{MAX}^2}{\mathrm{MSE}}\right)\\[3pt]
        \mathrm{SSIM}
          &=\frac{(2\mu_x\mu_y\!+\!c_1)(2\sigma_{xy}\!+\!c_2)}
                 {(\mu_x^2\!+\!\mu_y^2\!+\!c_1)(\sigma_x^2\!+\!\sigma_y^2\!+\!c_2)}
      \end{align*}
    \end{tcolorbox}
  \end{minipage}

  \vspace{10pt}
  % Middle — explanation
  \fontsize{26}{34}\selectfont\RaggedRight
  ALS iteratively updates each factor matrix while fixing the others, cycling
  until convergence. Three experiments validate correctness across real and
  complex tensors: (i)~exact rank-2 recovery, (ii)~noisy rank-2 recovery,
  (iii)~generic random tensor approximation.
  All cases satisfy $\varepsilon_{\mathrm{rel}}<10^{-4}$.
  \vspace{10pt}

  \fontsize{26}{34}\selectfont\RaggedRight
  \textbf{Key findings:}
  \begin{itemize}\setlength\itemsep{5pt}
    \item Tucker achieves higher PSNR than CP at the same compression ratio
    \item L1 (robust) objective significantly outperforms L2 under
          impulsive Salt-and-Pepper noise
    \item Rank selection is the dominant hyperparameter controlling the
